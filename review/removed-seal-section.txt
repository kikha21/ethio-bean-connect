======== the press-and-hold script ========

/* ============================================================
   9. THE INTERACTIVE MOMENT: press and hold to seal
   ============================================================ */
var sealBtn  = document.getElementById('sealbtn');
var slip     = document.getElementById('slip');
var sealHint = document.getElementById('sealhint');
var lotEl    = document.getElementById('lotno');
var hold = 0, holding = false, holdRaf = null, sealed = false, holdLast = 0;
var HOLD_MS = 900, RELEASE_MS = 550;   /* real time, so every refresh rate feels the same */

function sealTo(v){ if (!sealBtn) return; hold = clamp(v,0,1); sealBtn.style.setProperty('--hold', hold.toFixed(3)); }
function finishSeal(){
  if (sealed) return;
  sealed = true;
  sealBtn.classList.add('done');
  slip.classList.add('sealed');
  lotEl.textContent = 'EBC-' + (new Date()).getFullYear() + '-' + String(Math.floor(Math.random()*8999)+1000);
  sealHint.textContent = t('seal.done');
  sealHint.classList.add('seal-done');
}
/* it used to be dead after one use. Anyone who found it got exactly one go. */
function resetSeal(){
  sealed = false; holding = false; hold = 0; holdLast = 0;
  sealTo(0);
  sealBtn.classList.remove('done');
  slip.classList.remove('sealed');
  lotEl.textContent = '—';   /* placeholder glyph, matches the &mdash; in the markup */
  sealHint.textContent = t('seal.hint');
  sealHint.classList.remove('seal-done');
}
function holdLoop(now){
  var dt = Math.min(100, now - (holdLast || now));
  holdLast = now;
  hold += holding ? (dt / HOLD_MS) : -(dt / RELEASE_MS);
  sealTo(hold);
  if (hold >= 1){ finishSeal(); holdRaf = null; holdLast = 0; return; }
  if (hold <= 0 && !holding){ holdRaf = null; holdLast = 0; return; }
  holdRaf = requestAnimationFrame(holdLoop);
}
function startHold(e){
  if (sealed){ if (e && e.cancelable) e.preventDefault(); resetSeal(); return; }
  if (e && e.cancelable) e.preventDefault();
  holding = true;
  if (holdRaf === null) holdRaf = requestAnimationFrame(holdLoop);
}
function endHold(){ holding = false; if (holdRaf === null && hold > 0) holdRaf = requestAnimationFrame(holdLoop); }
sealBtn.addEventListener('mousedown', startHold);
sealBtn.addEventListener('touchstart', startHold, {passive:false});
addEventListener('mouseup', endHold);
addEventListener('touchend', endHold);
sealBtn.addEventListener('mouseleave', endHold);
sealBtn.addEventListener('keydown', function(e){ if (e.key === ' ' || e.key === 'Enter'){ e.preventDefault(); startHold(); } });
sealBtn.addEventListener('keyup', function(e){ if (e.key === ' ' || e.key === 'Enter') endHold(); });

/* ============================================================
   11. WHISPER PARTICLES
   ============================================================ */
(function(){
  var host = document.getElementById('motes');
  if (!host) return;
  var r = rng(11), html = '';
  for (var i=0;i<16;i++){
    html += '<i class="mote" style="left:' + (r()*100).toFixed(1) + '%;top:' + (r()*100).toFixed(1) + '%;'
         +  'animation-duration:' + (16 + r()*16).toFixed(1) + 's;animation-delay:-' + (r()*18).toFixed(1) + 's;'
         +  'opacity:' + (0.08 + r()*0.12).toFixed(2) + '"></i>';
  }
  host.innerHTML = html;
})();
